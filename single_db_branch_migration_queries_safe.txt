-- 5. Recreate branch RLS objects safely.
IF EXISTS (SELECT 1 FROM sys.security_policies WHERE name = 'BranchIsolationPolicy' AND schema_id = SCHEMA_ID('dbo'))
    DROP SECURITY POLICY dbo.BranchIsolationPolicy;
GO
IF OBJECT_ID('dbo.fn_branch_access', 'IF') IS NOT NULL
    DROP FUNCTION dbo.fn_branch_access;
GO
CREATE FUNCTION dbo.fn_branch_access(@branch_code NVARCHAR(100))
RETURNS TABLE
WITH SCHEMABINDING
AS
RETURN
    SELECT 1 AS fn_result
    WHERE @branch_code IS NULL
       OR @branch_code = CONVERT(NVARCHAR(100), SESSION_CONTEXT(N'branch_code'));
GO
CREATE SECURITY POLICY dbo.BranchIsolationPolicy
ADD FILTER PREDICATE dbo.fn_branch_access(branch_code) ON dbo.app_users,
ADD FILTER PREDICATE dbo.fn_branch_access(branch_code) ON dbo.app_settings,
ADD FILTER PREDICATE dbo.fn_branch_access(branch_code) ON dbo.staff_roles,
ADD FILTER PREDICATE dbo.fn_branch_access(branch_code) ON dbo.customers,
ADD FILTER PREDICATE dbo.fn_branch_access(branch_code) ON dbo.orders,
ADD FILTER PREDICATE dbo.fn_branch_access(branch_code) ON dbo.expenses,
ADD FILTER PREDICATE dbo.fn_branch_access(branch_code) ON dbo.tailors,
ADD FILTER PREDICATE dbo.fn_branch_access(branch_code) ON dbo.vehicles,
ADD FILTER PREDICATE dbo.fn_branch_access(branch_code) ON dbo.vehicle_repairs,
ADD FILTER PREDICATE dbo.fn_branch_access(branch_code) ON dbo.vendors,
ADD FILTER PREDICATE dbo.fn_branch_access(branch_code) ON dbo.vendor_purchase_orders,
ADD FILTER PREDICATE dbo.fn_branch_access(branch_code) ON dbo.customer_sell_requests,
ADD FILTER PREDICATE dbo.fn_branch_access(branch_code) ON dbo.vehicle_sale_orders
WITH (STATE = ON);
GO
