-- 2. Fix app_settings uniqueness for single shared DB.
-------------------------------------------------------------------------------
DECLARE @LegacyAppSettingsKeyConstraint SYSNAME;
DECLARE @DropAppSettingsConstraintSql NVARCHAR(MAX);

SELECT TOP 1 @LegacyAppSettingsKeyConstraint = kc.name
FROM sys.key_constraints kc
INNER JOIN sys.index_columns ic
    ON ic.object_id = kc.parent_object_id
   AND ic.index_id = kc.unique_index_id
INNER JOIN sys.columns c
    ON c.object_id = ic.object_id
   AND c.column_id = ic.column_id
WHERE kc.parent_object_id = OBJECT_ID('dbo.app_settings')
  AND kc.type = 'UQ'
GROUP BY kc.name
HAVING COUNT(*) = 1 AND MAX(c.name) = 'key';

IF @LegacyAppSettingsKeyConstraint IS NOT NULL
BEGIN
    SET @DropAppSettingsConstraintSql =
        N'ALTER TABLE dbo.app_settings DROP CONSTRAINT ' + QUOTENAME(@LegacyAppSettingsKeyConstraint) + N';';
    EXEC(@DropAppSettingsConstraintSql);
END

IF NOT EXISTS (
    SELECT 1
    FROM sys.indexes
    WHERE object_id = OBJECT_ID('dbo.app_settings')
      AND name = 'UQ_app_settings_key_branch_code'
)
BEGIN
    CREATE UNIQUE INDEX UQ_app_settings_key_branch_code
    ON dbo.app_settings([key], branch_code)
    WHERE branch_code IS NOT NULL;
END

IF NOT EXISTS (
    SELECT 1
    FROM sys.indexes
    WHERE object_id = OBJECT_ID('dbo.app_settings')
      AND name = 'UQ_app_settings_key_global'
)
BEGIN
    CREATE UNIQUE INDEX UQ_app_settings_key_global
    ON dbo.app_settings([key])
    WHERE branch_code IS NULL;
END
GO

-- 3. Fix vehicles uniqueness for single shared DB.
-------------------------------------------------------------------------------
DECLARE @LegacyVehiclesNumberConstraint SYSNAME;
DECLARE @DropVehiclesConstraintSql NVARCHAR(MAX);

SELECT TOP 1 @LegacyVehiclesNumberConstraint = kc.name
FROM sys.key_constraints kc
INNER JOIN sys.index_columns ic
    ON ic.object_id = kc.parent_object_id
   AND ic.index_id = kc.unique_index_id
INNER JOIN sys.columns c
    ON c.object_id = ic.object_id
   AND c.column_id = ic.column_id
WHERE kc.parent_object_id = OBJECT_ID('dbo.vehicles')
  AND kc.type = 'UQ'
GROUP BY kc.name
HAVING COUNT(*) = 1 AND MAX(c.name) = 'vehicle_number';

IF @LegacyVehiclesNumberConstraint IS NOT NULL
BEGIN
    SET @DropVehiclesConstraintSql =
        N'ALTER TABLE dbo.vehicles DROP CONSTRAINT ' + QUOTENAME(@LegacyVehiclesNumberConstraint) + N';';
    EXEC(@DropVehiclesConstraintSql);
END

IF NOT EXISTS (
    SELECT 1
    FROM sys.indexes
    WHERE object_id = OBJECT_ID('dbo.vehicles')
      AND name = 'UQ_vehicles_vehicle_number_branch_code'
)
BEGIN
    CREATE UNIQUE INDEX UQ_vehicles_vehicle_number_branch_code
    ON dbo.vehicles(vehicle_number, branch_code)
    WHERE branch_code IS NOT NULL AND vehicle_number IS NOT NULL;
END

IF NOT EXISTS (
    SELECT 1
    FROM sys.indexes
    WHERE object_id = OBJECT_ID('dbo.vehicles')

END

-------------------------------------------------------------------------------
-- 7. Row-level security function.
-------------------------------------------------------------------------------
CREATE FUNCTION dbo.fn_branch_access(@branch_code NVARCHAR(100))
RETURNS TABLE
WITH SCHEMABINDING
AS
RETURN
    SELECT 1 AS fn_result
    WHERE @branch_code IS NULL
       OR @branch_code = CONVERT(NVARCHAR(100), SESSION_CONTEXT(N'branch_code'));
GO
